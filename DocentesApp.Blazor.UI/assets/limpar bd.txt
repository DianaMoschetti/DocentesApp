-- sacar materias
SELECT a.Id, a.Nombre, a.Frecuencia, a.Nivel, a.EsVigente, a.UdbId, u.Nombre as NombreUdb
FROM Asignaturas a
JOIN Udbs u ON a.UdbId = u.Id
ORDER BY u.Nombre, a.Nivel, a.Nombre
